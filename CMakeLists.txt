cmake_minimum_required(VERSION 3.21)
project(SolidVolumeLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(volume volume.cpp scene.cpp map_source.cpp map_world.cpp)
add_executable(map-world-tests map_world_tests.cpp)
target_link_libraries(map-world-tests PRIVATE volume)
add_executable(map-source-tests map_source_tests.cpp)
target_link_libraries(map-source-tests PRIVATE volume)
add_executable(volume-tests tests.cpp)
target_link_libraries(volume-tests PRIVATE volume)
enable_testing()
add_test(NAME map-source-contract COMMAND map-source-tests)
add_test(NAME map-world-contract COMMAND map-world-tests)
add_test(NAME volume-contract COMMAND volume-tests)
if(WIN32)
  add_executable(DERP-SolidVolume-Lab WIN32 viewer.cpp)
  target_compile_definitions(DERP-SolidVolume-Lab PRIVATE NOMINMAX)
  target_link_libraries(DERP-SolidVolume-Lab PRIVATE volume opengl32 glu32 gdi32 user32)
endif()
